#!/usr/bin/env python3
# ~/.pigo/plugins/wordcount.py   (chmod +x)
import json, sys

def reply(id_, result):
    print(json.dumps({"jsonrpc": "2.0", "id": id_, "result": result}), flush=True)

for line in sys.stdin:
    line = line.strip()
    if not line:
        continue
    req = json.loads(line)
    method, rid = req.get("method"), req.get("id")

    if method == "initialize":
        reply(rid, {
            "name": "wordcount",
            "version": "0.1.0",
            "tools": [{
                "name": "word_count",
                "description": "Count words in a piece of text",
                "schema": {
                    "type": "object",
                    "properties": {"text": {"type": "string"}},
                    "required": ["text"]
                }
            }]
        })

    elif method == "tools/call":
        args = req["params"]["arguments"]
        text = args.get("text", "")
        reply(rid, {"content": str(len(text.split()))})

    elif method == "shutdown":
        break

#!/usr/bin/env python3
# ~/.pigo/plugins/demo.py   (chmod +x)
import json, sys

def send(obj):
    print(json.dumps(obj), flush=True)

def reply(rid, result):
    send({"jsonrpc": "2.0", "id": rid, "result": result})

def log(msg):
    print(f"[demo] {msg}", file=sys.stderr, flush=True)   # stderr is for logs

for line in sys.stdin:
    line = line.strip()
    if not line:
        continue
    req = json.loads(line)
    method, rid = req.get("method"), req.get("id")

    if method == "initialize":
        reply(rid, {
            "name": "demo",
            "version": "0.1.0",
            "tools": [{
                "name": "demo_echo",
                "description": "Echo the input text",
                "schema": {"type": "object", "properties": {"text": {"type": "string"}}, "required": ["text"]}
            }],
            "commands": [{"name": "demo", "description": "Print demo status"}],
            "events": ["turn_end"]
        })

    elif method == "tools/call":
        text = req["params"]["arguments"].get("text", "")
        reply(rid, {"content": text})

    elif method == "commands/call":
        reply(rid, {
            "notifications": [{"message": "demo plugin is alive", "type": "info"}],
            "prompt": "Continue with the user's original request."
        })

    elif method == "event":
        ev = req["params"]
        if ev["type"] == "turn_end":
            log(f"turn ended: {ev.get('data', {})}")
        # notifications must not be answered

    elif method == "shutdown":
        log("bye")
        break
